            print(y, "<", x)

10232 - باقر خسته‌ست ولی بی‌فرهنگ نه

کد 
"""
----input
2 10
3 5 5
5 2 2
----output
12
----input
4 30
7 13 5
14 4 4
15 3 10
25 1 1
----output
36
"""
n, l = [int(i) for i in input().split(" ")]
time = 0
last_redlight = 0

for i in range(n):
    d, r, g = [int(i) for i in input().split(" ")]

    time +=  d - last_redlight

    if time % (r + g) < r:
        time += r - time % (r + g)
    
    last_redlight = d

time += l - last_redlight
print(time)

148640 - آزمون تستی

کد 
"""
----input
1
C
4
OO#O
OOOO
#OOO
OO##
----output
3
0
-1
-1
----input
10
AABBDCABCD
1
#OOO
#OOO
O#OO
O#OO
OOO#
OO#O
#OOO
O#OO
OO#O
OOO#
----output
30
"""
def check_score(answer, response):
    abcd = {"A": 0, "B": 1, "C": 2, "D": 3}
    score = 0
    if "#" not in response:
        return 0

    for i in range(4):
        if response[i] == "#":
            if i != abcd[answer]:
                return -1
            else:
                score = 3

    return score

questions = int(input())
answers = input()
papers = int(input())


for p in range(papers):
    score = 0
    for q in range(questions):
        response = input()
        score += check_score(answers[q], response)
    
    print(score)
